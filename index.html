import os import time import sqlite3 import asyncio import logging from datetime import datetime, timezone import requests from fastapi import FastAPI, HTTPException from fastapi.responses import HTMLResponse from pydantic import BaseModel # ===================================================== # تنظیمات MAHORA AI # ===================================================== APP_NAME = "MAHORA AI" DB_FILE = os.getenv("DB_FILE", "mahora.db") INITIAL_BALANCE = float(os.getenv("PAPER_BALANCE", "10000000")) USDT_TOMAN = float(os.getenv("USDT_TOMAN", "100000")) FEE_RATE = 0.001 SLIPPAGE = 0.0005 RISK_PER_TRADE = 0.01 MAX_OPEN_TRADES = 3 MIN_AGREEMENT = 6 TIMEFRAME = "15m" LOOP_SECONDS = 60 SYMBOLS = [ "BTCUSDT", "ETHUSDT", "XRPUSDT", "BNBUSDT", "SOLUSDT", "DOGEUSDT", "PEPEUSDT" ] logging.basicConfig(level=logging.INFO) log = logging.getLogger("MAHORA") app = FastAPI( title="MAHORA AI", version="1.0.0", description="سامانه معاملات آزمایشی ماهورا" ) # ===================================================== # دیتابیس # ===================================================== def db(): con = sqlite3.connect(DB_FILE, timeout=20) con.row_factory = sqlite3.Row return con def init_db(): con = db() cur = con.cursor() cur.execute(""" CREATE TABLE IF NOT EXISTS settings ( id INTEGER PRIMARY KEY CHECK(id=1), cash REAL NOT NULL, initial_balance REAL NOT NULL, enabled INTEGER NOT NULL DEFAULT 0, emergency INTEGER NOT NULL DEFAULT 0 ) """) cur.execute(""" CREATE TABLE IF NOT EXISTS trades ( id INTEGER PRIMARY KEY AUTOINCREMENT, symbol TEXT NOT NULL, entry REAL NOT NULL, quantity REAL NOT NULL, stop REAL NOT NULL, target REAL NOT NULL, highest REAL NOT NULL, entry_fee REAL NOT NULL, status TEXT NOT NULL, exit REAL, exit_fee REAL DEFAULT 0, pnl REAL DEFAULT 0, reason TEXT, opened_at TEXT NOT NULL, closed_at TEXT ) """) cur.execute(""" CREATE TABLE IF NOT EXISTS signals ( id INTEGER PRIMARY KEY AUTOINCREMENT, symbol TEXT NOT NULL, price REAL NOT NULL, direction TEXT NOT NULL, agreement INTEGER NOT NULL, strategies TEXT NOT NULL, created_at TEXT NOT NULL ) """) cur.execute(""" INSERT OR IGNORE INTO settings (id, cash, initial_balance, enabled, emergency) VALUES (1, ?, ?, 0, 0) """, (INITIAL_BALANCE, INITIAL_BALANCE)) con.commit() con.close() def get_settings(): con = db() row = con.execute( "SELECT * FROM settings WHERE id=1" ).fetchone() con.close() return dict(row) def update_settings(**values): allowed = {"cash", "enabled", "emergency"} fields = {k: v for k, v in values.items() if k in allowed} if not fields: return con = db() sql = ", ".join(f"{k}=?" for k in fields) con.execute( f"UPDATE settings SET {sql} WHERE id=1", list(fields.values()) ) con.commit() con.close() # ===================================================== # دریافت اطلاعات بازار # ===================================================== BINANCE_URL = "https://api.binance.com/api/v3/klines" def get_candles(symbol): response = requests.get( BINANCE_URL, params={ "symbol": symbol, "interval": TIMEFRAME, "limit": 150 }, timeout=12, headers={"User-Agent": "MAHORA-AI/1.0"} ) response.raise_for_status() raw = response.json() # کندل آخر هنوز ممکن است بسته نشده باشد. raw = raw[:-1] return [{ "time": int(x[0]), "open": float(x[1]), "high": float(x[2]), "low": float(x[3]), "close": float(x[4]), "volume": float(x[5]) } for x in raw] # ===================================================== # اندیکاتورها # ===================================================== def ema(values, period): if len(values) < period: return None result = sum(values[:period]) / period alpha = 2 / (period + 1) for value in values[period:]: result = value * alpha + result * (1 - alpha) return result def sma(values, period): if len(values) < period: return None return sum(values[-period:]) / period def rsi(values, period=14): if len(values) <= period: return None changes = [ values[i] - values[i - 1] for i in range(1, len(values)) ] gains = [max(x, 0) for x in changes[-period:]] losses = [max(-x, 0) for x in changes[-period:]] avg_gain = sum(gains) / period avg_loss = sum(losses) / period if avg_loss == 0: return 100 rs = avg_gain / avg_loss return 100 - (100 / (1 + rs)) def atr(candles, period=14): if len(candles) <= period: return None ranges = [] for i in range(1, len(candles)): c = candles[i] previous = candles[i - 1]["close"] ranges.append(max( c["high"] - c["low"], abs(c["high"] - previous), abs(c["low"] - previous) )) return sum(ranges[-period:]) / period def macd(values): fast = ema(values, 12) slow = ema(values, 26) if fast is None or slow is None: return None return fast - slow def bollinger(values, period=20): if len(values) < period: return None, None sample = values[-period:] middle = sum(sample) / period variance = sum( (x - middle) ** 2 for x in sample ) / period deviation = variance ** 0.5 return middle + 2 * deviation, middle - 2 * deviation def stochastic(candles, period=14): sample = candles[-period:] highest = max(x["high"] for x in sample) lowest = min(x["low"] for x in sample) close = candles[-1]["close"] if highest == lowest: return 50 return 100 * (close - lowest) / (highest - lowest) # ===================================================== # ده استراتژی تحلیلی # ===================================================== def analyze(candles): closes = [x["close"] for x in candles] volumes = [x["volume"] for x in candles] price = closes[-1] votes = [] e9 = ema(closes, 9) e21 = ema(closes, 21) e50 = ema(closes, 50) e100 = ema(closes, 100) r = rsi(closes) m = macd(closes) upper, lower = bollinger(closes) stoch = stochastic(candles) # 1. تقاطع میانگین نمایی if e9 and e21: votes.append(("تقاطع EMA", 1 if e9 > e21 else -1)) # 2. روند EMA50 if e50: votes.append(("روند EMA50", 1 if price > e50 else -1)) # 3. روند EMA100 if e100: votes.append(("روند EMA100", 1 if price > e100 else -1)) # 4. شاخص RSI if r is not None: votes.append(( "شاخص RSI", 1 if 50 < r < 70 else (-1 if r > 70 or r < 40 else 0) )) # 5. مکدی if m is not None: votes.append(("MACD", 1 if m > 0 else -1)) # 6. شکست سقف و کف recent_high = max(x["high"] for x in candles[-21:-1]) recent_low = min(x["low"] for x in candles[-21:-1]) votes.append(( "شکست قیمت", 1 if price > recent_high else ( -1 if price < recent_low else 0 ) )) # 7. باند بولینگر if upper and lower: votes.append(( "بولینگر", 1 if price < lower else ( -1 if price > upper else 0 ) )) # 8. استوکاستیک votes.append(( "استوکاستیک", 1 if stoch < 20 else (-1 if stoch > 80 else 0) )) # 9. مومنتوم momentum = closes[-1] - closes[-6] votes.append(( "مومنتوم", 1 if momentum > 0 else (-1 if momentum < 0 else 0) )) # 10. حجم معاملات average_volume = sma(volumes[:-1], 20) if average_volume and average_volume > 0: volume_signal = ( 1 if volumes[-1] > average_volume * 1.5 and closes[-1] > candles[-2]["close"] else 0 ) votes.append(("حجم معاملات", volume_signal)) buy_count = sum(1 for _, v in votes if v == 1) sell_count = sum(1 for _, v in votes if v == -1) direction = "WAIT" if buy_count >= MIN_AGREEMENT and buy_count > sell_count: direction = "BUY" elif sell_count >= MIN_AGREEMENT and sell_count > buy_count: direction = "SELL" return { "price": price, "direction": direction, "buy_votes": buy_count, "sell_votes": sell_count, "agreement": max(buy_count, sell_count), "strategies": [ name for name, vote in votes if vote == (1 if direction == "BUY" else -1) ] if direction != "WAIT" else [], "atr": atr(candles), "rsi": r, "stoch": stoch } # ===================================================== # موتور معاملات آزمایشی # ===================================================== def open_positions(): con = db() rows = con.execute( "SELECT * FROM trades WHERE status='OPEN'" ).fetchall() con.close() return [dict(x) for x in rows] def log_signal(symbol, result): con = db() con.execute(""" INSERT INTO signals (symbol, price, direction, agreement, strategies, created_at) VALUES (?, ?, ?, ?, ?, ?) """, ( symbol, result["price"], result["direction"], result["agreement"], "، ".join(result["strategies"]), datetime.now(timezone.utc).isoformat() )) con.commit() con.close() def paper_buy(symbol, result): settings = get_settings() positions = open_positions() if len(positions) >= MAX_OPEN_TRADES: return if any(p["symbol"] == symbol for p in positions): return price = result["price"] volatility = result["atr"] if not volatility or volatility <= 0: return entry = price * (1 + SLIPPAGE) stop = entry - 1.5 * volatility target = entry + 2.2 * volatility if stop <= 0 or target <= entry: return risk_budget = settings["cash"] * RISK_PER_TRADE risk_per_unit = (entry - stop) * USDT_TOMAN if risk_per_unit <= 0: return quantity_by_risk = risk_budget / risk_per_unit available_usdt = settings["cash"] / USDT_TOMAN quantity_by_cash = available_usdt / ( entry * (1 + FEE_RATE) ) quantity = min(quantity_by_risk, quantity_by_cash) if quantity <= 0: return notional_toman = quantity * entry * USDT_TOMAN entry_fee = notional_toman * FEE_RATE total_cost = notional_toman + entry_fee if total_cost > settings["cash"]: return con = db() con.execute(""" INSERT INTO trades ( symbol, entry, quantity, stop, target, highest, entry_fee, status, opened_at ) VALUES (?, ?, ?, ?, ?, ?, ?, 'OPEN', ?) """, ( symbol, entry, quantity, stop, target, entry, entry_fee, datetime.now(timezone.utc).isoformat() )) con.execute( "UPDATE settings SET cash=cash-? WHERE id=1", (total_cost,) ) con.commit() con.close() def close_trade(trade, price, reason): exit_price = price * (1 - SLIPPAGE) proceeds = ( trade["quantity"] * exit_price * USDT_TOMAN ) exit_fee = proceeds * FEE_RATE entry_cost = ( trade["entry"] * trade["quantity"] * USDT_TOMAN ) pnl = ( proceeds - exit_fee - entry_cost - trade["entry_fee"] ) con = db() con.execute(""" UPDATE trades SET status='CLOSED', exit=?, exit_fee=?, pnl=?, reason=?, closed_at=? WHERE id=? AND status='OPEN' """, ( exit_price, exit_fee, pnl, reason, datetime.now(timezone.utc).isoformat(), trade["id"] )) con.execute( "UPDATE settings SET cash=cash+? WHERE id=1", (proceeds - exit_fee,) ) con.commit() con.close() def manage_trade(trade, result): price = result["price"] volatility = result["atr"] if not volatility or volatility <= 0: return # فقط افزایش حد ضرر در معامله خرید highest = max(trade["highest"], price) trailing_stop = highest - 1.5 * volatility new_stop = max(trade["stop"], trailing_stop) con = db() con.execute( "UPDATE trades SET highest=?, stop=? WHERE id=?", (highest, new_stop, trade["id"]) ) con.commit() con.close() trade["stop"] = new_stop if price <= new_stop: close_trade(trade, price, "حد ضرر پویا") elif price >= trade["target"]: close_trade(trade, price, "حد سود") elif result["direction"] == "SELL" and result["agreement"] >= 8: close_trade(trade, price, "تغییر قوی سیگنال") # ===================================================== # چرخه تحلیل بازار # ===================================================== market_cache = {} signal_cache = {} last_candle = {} def process_market(): settings = get_settings() if settings["emergency"]: return for symbol in SYMBOLS: try: candles = get_candles(symbol) if len(candles) < 110: continue result = analyze(candles) market_cache[symbol] = { "price": result["price"], "updated": datetime.now(timezone.utc).isoformat() } signal_cache[symbol] = result log_signal(symbol, result) positions = open_positions() for trade in positions: if trade["symbol"] == symbol: manage_trade(trade, result) settings = get_settings() if not settings["enabled"] or settings["emergency"]: continue # فقط یک ورود در هر کندل بسته‌شده candle_time = candles[-1]["time"] if last_candle.get(symbol) == candle_time: continue if result["direction"] == "BUY": paper_buy(symbol, result) last_candle[symbol] = candle_time except Exception as exc: log.warning("%s market error: %s", symbol, exc) async def engine_loop(): while True: try: await asyncio.to_thread(process_market) except Exception: log.exception("Engine cycle failed") await asyncio.sleep(LOOP_SECONDS) @app.on_event("startup") async def start_engine(): init_db() asyncio.create_task(engine_loop()) # ===================================================== # API حساب و بازار # ===================================================== @app.get("/api/status") def status(): settings = get_settings() positions = open_positions() return { "name": APP_NAME, "enabled": bool(settings["enabled"]), "emergency": bool(settings["emergency"]), "mode": "PAPER", "real_trading": False, "cash_toman": round(settings["cash"], 2), "initial_toman": settings["initial_balance"], "open_positions": len(positions), "symbols": SYMBOLS, "timeframe": TIMEFRAME } @app.get("/api/market") def market(): return { "market": market_cache, "signals": signal_cache } @app.get("/api/trades") def trades(): con = db() rows = con.execute( "SELECT * FROM trades ORDER BY id DESC LIMIT 200" ).fetchall() con.close() return [dict(x) for x in rows] @app.get("/api/summary") def summary(): con = db() closed = con.execute(""" SELECT COUNT(*) AS total, COALESCE(SUM(pnl),0) AS pnl, SUM(CASE WHEN pnl > 0 THEN 1 ELSE 0 END) AS wins FROM trades WHERE status='CLOSED' """).fetchone() opened = con.execute(""" SELECT COUNT(*) AS total FROM trades WHERE status='OPEN' """).fetchone() con.close() total = closed["total"] or 0 wins = closed["wins"] or 0 return { "closed_trades": total, "open_trades": opened["total"], "net_pnl_toman": round(closed["pnl"], 2), "win_rate_percent": round( wins / total * 100, 2 ) if total else 0 } #
